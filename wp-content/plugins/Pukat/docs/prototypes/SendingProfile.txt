    function openSmtpSlideover() {
      const container = document.getElementById('smtp-slideover-container');
      if (container) {
        container.classList.remove('hidden');
        setTimeout(() => {
          container.classList.add('slideover-active');
          container.classList.add('active');
          container.style.opacity = '1';
        }, 50);
      }
    }

    function closeSmtpSlideover() {
      const container = document.getElementById('smtp-slideover-container');
      if (container) {
        container.classList.remove('slideover-active');
        container.classList.remove('active');
        container.style.opacity = '0';
        setTimeout(() => {

    function filterSmtpTable(query) {
      const q = query.toLowerCase();
      const filtered = smtpProfiles.filter(p => 
        p.name.toLowerCase().includes(q) || 
        p.host.toLowerCase().includes(q) || 
        p.from.toLowerCase().includes(q) || 
        p.encryption.toLowerCase().includes(q)
      );
      renderSmtpTable(filtered);
    }

    function updateSmtpPortDefault(enc) {
      const portInput = document.getElementById('smtp-port');
      if (portInput) {
        if (enc === 'TLS') portInput.value = '587';
        else if (enc === 'SSL') portInput.value = '465';
        else if (enc === 'None') portInput.value = '25';
      }
    }

      const pill = document.getElementById('smtp-change-pill');
      if (pill) {

    function addSmtpHeaderField(key = '', val = '') {
      const container = document.getElementById('smtp-headers-container');

      if (isNaN(idx) || idx === -1) return;

        renderSmtpTable();
        showToast(`Profil SMTP '${prof.name}' berhasil dihapus.`, 'success');
        closeSmtpSlideover();
      }
    }

    function submitSmtpProfile() {
      const mode = document.getElementById('smtp-edit-mode').value;
      const name = document.getElementById('smtp-name').value.trim();
      const host = document.getElementById('smtp-host').value.trim();
      const port = parseInt(document.getElementById('smtp-port').value);
      const enc = document.getElementById('smtp-encryption').value;
      const user = document.getElementById('smtp-user').value.trim();
      const pass = document.getElementById('smtp-pass').value;
      const from = document.getElementById('smtp-from').value.trim();
      
      const ignoreToggle = document.getElementById('smtp-ignore-toggle');
      const ignoreCert = ignoreToggle ? !ignoreToggle.classList.contains('off') : false;

      if (!name || !host || isNaN(port) || !from) {
        showToast('Mohon lengkapi field wajib (*)', 'error');
        return;
      }

      // Collect headers
      const headers = [];
      const headerRows = document.querySelectorAll('#smtp-headers-container .header-row');
      headerRows.forEach(row => {
        const k = row.querySelector('.header-key').value.trim();
        const v = row.querySelector('.header-val').value.trim();
        if (k) {
          headers.push({ key: k, val: v });
        }
      });

      if (mode === 'new' || mode === 'dup') {
        // Check duplicate name
        if (smtpProfiles.some(p => p.name.toLowerCase() === name.toLowerCase())) {
          showToast(`Nama profil '${name}' sudah digunakan.`, 'error');
          return;
        }

        const newProf = {
          name: name,
          host: host,
          port: port,
          from: from,
          encryption: enc,
          status: 'Valid',
          used: '0 playbooks',
          lastTest: 'Baru saja',
          username: user,
          password: pass,
          ignoreCert: ignoreCert,
          headers: headers
        };
        smtpProfiles.push(newProf);
        showToast(`Profil SMTP '${name}' berhasil dibuat.`, 'success');
      } else if (mode === 'update') {
        const idx = parseInt(document.getElementById('smtp-edit-index').value);
        if (isNaN(idx) || idx === -1) return;

        // Check duplicate name excluding itself
        if (smtpProfiles.some((p, i) => i !== idx && p.name.toLowerCase() === name.toLowerCase())) {
          showToast(`Nama profil '${name}' sudah digunakan.`, 'error');
          return;
        }

        const prof = smtpProfiles[idx];
        prof.name = name;
        prof.host = host;
        prof.port = port;
        prof.encryption = enc;
        prof.from = from;
        prof.username = user;
        prof.password = pass;
        prof.ignoreCert = ignoreCert;
        prof.headers = headers;
        prof.lastTest = 'Baru saja';
        prof.status = 'Valid';

        showToast(`Profil SMTP '${name}' berhasil disimpan.`, 'success');
      }

      renderSmtpTable();
      closeSmtpSlideover();
    }
